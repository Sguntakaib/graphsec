```typescript
class NodeIntelligence {
  onNodeAdded(node: SecurityNode): void {
    const prompts = this.getSecurityPrompts(node.subtype);
    this.showSecurityQuestionnaire(prompts);
    this.createRequiredBranches(node);
  }
  
  getSecurityPrompts(nodeType: string): SecurityPrompt[] {
    switch(nodeType) {
      case 'Website':
        return [
          { question: "Authentication type?", options: ["Password", "SSO", "MFA"] },
          { question: "API endpoints present?", type: "boolean" },
          { question: "Database connection?", type: "boolean" }
        ];
    }
  }
}
```

🔧 **3. DSL Rule Engine**
```python
class ThreatRuleEngine:
    def evaluate_threats(self, graph: SecurityGraph) -> List[ThreatResult]:
        results = []
        for rule in self.rules:
            if self.matches_conditions(rule.when, graph):
                threat = self.calculate_threat(rule, graph)
                results.append(threat)
        return results
    
    def calculate_threat_probability(self, path: AttackPath) -> float:
        probability = 1.0
        for edge in path.edges:
            vuln_factor = edge.vulnerability_likelihood
            control_coverage = edge.control_effectiveness
            probability *= vuln_factor * (1 - control_coverage)
        return probability
```

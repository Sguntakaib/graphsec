// ADD: Intelligent next question logic

const SecurityQuestionnaire = ({ node, onClose, onComplete }) => {
  // EXISTING: All Phase 1 state - keep unchanged
  
  // NEW: Add question flow state
  const [questionFlow, setQuestionFlow] = useState([]);
  const [estimatedProgress, setEstimatedProgress] = useState(0);
  
  // ENHANCE: Question navigation with branching
  const getNextQuestionIndex = (currentIndex, context) => {
    if (!isDynamic) {
      // EXISTING: Linear progression - unchanged
      return currentIndex + 1;
    }
    
    // NEW: Dynamic next question logic
    const currentQuestion = questions[currentIndex];
    const nextIndex = findNextRelevantQuestion(
      questions, 
      currentIndex + 1, 
      context, 
      responses
    );
    
    return nextIndex;
  };
  
  // NEW: Intelligent question filtering
  const findNextRelevantQuestion = (questions, startIndex, context, responses) => {
    for (let i = startIndex; i < questions.length; i++) {
      const question = questions[i];
      
      // Skip questions based on context
      if (shouldSkipQuestion(question.id, context)) {
        continue;
      }
      
      // Skip if dependencies not met
      if (!areDependenciesMet(question, responses)) {
        continue;
      }
      
      return i;
    }
    
    return questions.length; // End of questionnaire
  };
  
  // NEW: Progress estimation based on context
  const calculateProgress = (context, responses) => {
    if (!isDynamic) {
      // EXISTING calculation - unchanged
      return (Object.keys(responses).length / questions.length) * 100;
    }
    
    // NEW: Context-aware progress
    const relevantQuestions = questions.filter(q => 
      !shouldSkipQuestion(q.id, context)
    );
    const answeredRelevant = Object.keys(responses).filter(qId =>
      relevantQuestions.find(q => q.id === qId)
    );
    
    return (answeredRelevant.length / relevantQuestions.length) * 100;
  };
};

class VulnerabilityEngine:
    # EXISTING: All current methods - keep unchanged
    
    # NEW: Context-aware vulnerability analysis (enhance existing method)

        # EXISTING: Base vulnerability analysis - keep unchanged

        # NEW: Apply context filtering (if context provided)

        app_type = context.get('app_type')
        
        # Filter out irrelevant vulnerabilities
        relevant_vulns = []
        for vuln in vulnerabilities:
            if self.is_vulnerability_relevant(vuln, context):
                # Adjust severity based on context
                vuln.severity = self.adjust_severity_by_context(vuln, context)
                relevant_vulns.append(vuln)
        
        return relevant_vulns
    
    # NEW: Relevance checking (20 lines)
    def is_vulnerability_relevant(self, vulnerability, context: Dict) -> bool:
        app_type = context.get('app_type')
        
        # Static sites don't have authentication vulnerabilities
        if app_type == 'static' and 'authentication' in vulnerability.category.lower():
            return False
            
        # SaaS platforms need multi-tenant specific checks
        if app_type == 'saas' and vulnerability.category == 'multi_tenant':
            return True
            
        return True  # Default: include vulnerability

#### **2.3 Enhance Existing STRIDE Engine (Day 3-4)**

```python
# EXISTING: stride_engine.py  
# ENHANCE: Add context awareness (reuse 90% of existing code)

class StrideRuleEngine:
    # EXISTING: All current STRIDE rules - keep unchanged
    
    # ENHANCE: Existing analyze_node_threats method
    def analyze_node_threats(self, node: Dict, questionnaire_responses: Dict, context: Dict = None) -> List[Threat]:
        # EXISTING: Base STRIDE analysis - keep unchanged
        base_threats = self.get_base_threats(node, questionnaire_responses)
        
        # NEW: Apply context filtering (if available)
        if context:
            filtered_threats = self.filter_threats_by_context(base_threats, context)
            return filtered_threats
            
        return base_threats
    
    # NEW: Context-based threat filtering (25 lines)
    def filter_threats_by_context(self, threats: List[Threat], context: Dict) -> List[Threat]:
        app_type = context.get('app_type')
        relevant_threats = []
        
        for threat in threats:
            # Skip spoofing for static sites (no authentication)
            if app_type == 'static' and threat.stride_category == StrideCategory.SPOOFING:
                continue
                
            # Adjust threat severity based on context
            threat.residual_risk = self.adjust_risk_by_context(threat, context)
            relevant_threats.append(threat)
            
        return relevant_threats
    
    # NEW: Context-based risk adjustment (15 lines)  
    def adjust_risk_by_context(self, threat: Threat, context: Dict) -> float:
        base_risk = threat.residual_risk
        
        # Increase risk for SaaS multi-tenant issues
        if context.get('multi_tenant') and 'isolation' in threat.description.lower():
            return min(base_risk * 1.5, 10.0)
            
        # Decrease risk for static sites
        if context.get('app_type') == 'static':
            return base_risk * 0.7
            
        return base_risk
```

## 📊 **EXECUTIVE SUMMARY**

### **System Goals**
- **70% reduction in irrelevant questions** through context-aware filtering
- **40% increase in threat accuracy** via intelligent branching
- **Real-time adaptability** based on user responses
- **Template-based reusability** for common configurations
- **Cross-node dependency management** with intelligent suggestions

### **Key Architectural Innovations**
1. **Dynamic Question Engine**: Conditional branching based on context
2. **Context-Aware Vulnerability Analysis**: Application-type specific threat detection
3. **Intelligent STRIDE Mapping**: Relevance-filtered threat modeling
4. **Template Management System**: Save/load/share configurations
5. **Dependency Graph Engine**: Auto-suggest related components

#### **Backend Components**
- **Framework**: FastAPI (Python 3.11+)
- **Database**: MongoDB with aggregation pipelines
- **Caching**: Redis for question flow caching
- **Queue**: Celery for async template processing
- **Validation**: Pydantic v2 for data validation

#### **Frontend Components**  
- **Framework**: React 19 with Context API
- **State Management**: Zustand for complex flows
- **UI Library**: Radix UI + Tailwind CSS
- **Forms**: React Hook Form with Zod validation
- **Flow Visualization**: React Flow for dependency mapping

### **Core Backend Services**

#### **1. Dynamic Question Engine**

```python
from typing import Dict, List, Optional, Any
from pydantic import BaseModel
from enum import Enum

class QuestionType(str, Enum):
    SINGLE_CHOICE = "single_choice"
    MULTIPLE_CHOICE = "multiple_choice" 
    TEXT_INPUT = "text_input"
    NODE_SELECTOR = "node_selector"
    BOOLEAN = "boolean"

class QuestionOption(BaseModel):
    value: str
    label: str
    context_updates: Dict[str, Any] = {}
    next_questions: List[str] = []
    skip_question_categories: List[str] = []
    dependency_trigger: Optional[Dict[str, Any]] = None
    condition: Optional[str] = None

class DynamicQuestion(BaseModel):
    id: str
    question: str
    type: QuestionType
    required: bool = True
    options: List[QuestionOption] = []
    context_dependent: bool = False
    help_text: Optional[str] = None
    validation_rules: Dict[str, Any] = {}

class QuestionnaireSession(BaseModel):
    session_id: str
    node_id: str
    diagram_id: str
    current_flow: str
    current_question_id: Optional[str]
    context: Dict[str, Any] = {}
    responses: Dict[str, Any] = {}
    completed_questions: List[str] = []
    progress_percentage: float = 0.0
    estimated_remaining: int = 0

class DynamicQuestionEngine:
    def __init__(self, db_client, context_analyzer, dependency_engine):
        self.db = db_client
        self.context_analyzer = context_analyzer
        self.dependency_engine = dependency_engine
        
    async def start_session(self, node_type: str, diagram_id: str, node_id: str) -> QuestionnaireSession:
        """Initialize new dynamic questionnaire session"""
        
        # Get initial question for node type
        questionnaire_def = await self.get_questionnaire_definition(node_type)
        initial_question = questionnaire_def["schema"]["initial_question"]
        
        session = QuestionnaireSession(
            session_id=generate_uuid(),
            node_id=node_id,
            diagram_id=diagram_id,
            current_flow="initial",
            current_question_id=initial_question["id"],
            context={},
            estimated_remaining=self.estimate_questions(node_type, {})
        )
        
        # Store session in database
        await self.save_session(session)
        return session
        
    async def process_response(self, session_id: str, question_id: str, response: Any) -> Dict[str, Any]:
        """Process user response and determine next question"""
        
        session = await self.get_session(session_id)
        questionnaire_def = await self.get_questionnaire_definition_by_session(session)
        
        # Update session with response
        session.responses[question_id] = response
        session.completed_questions.append(question_id)
        
        # Update context based on response
        updated_context = await self.update_context(session, question_id, response, questionnaire_def)
        session.context.update(updated_context)
        
        # Check for dependency triggers
        dependencies = await self.check_dependency_triggers(session, question_id, response, questionnaire_def)
        
        # Determine next question using branching logic
        next_question = await self.determine_next_question(session, questionnaire_def)
        
        # Update progress
        session.progress_percentage = self.calculate_progress(session)
        session.estimated_remaining = self.estimate_remaining_questions(session)
        
        if next_question:
            session.current_question_id = next_question["id"]
            session.current_flow = next_question.get("flow", session.current_flow)
        else:
            # Questionnaire complete
            session.current_question_id = None
            session.progress_percentage = 100.0
            
        await self.save_session(session)
        
        return {
            "next_question": next_question,
            "context_updates": updated_context,
            "suggested_dependencies": dependencies,
            "progress": {
                "percentage": session.progress_percentage,
                "completed": len(session.completed_questions),
                "estimated_remaining": session.estimated_remaining
            },
            "session_complete": next_question is None
        }
        
    async def determine_next_question(self, session: QuestionnaireSession, questionnaire_def: Dict) -> Optional[DynamicQuestion]:
        """Intelligent next question determination with branching logic"""
        
        current_response = session.responses.get(session.current_question_id)
        if not current_response:
            return None
            
        # Get current question definition
        current_question = await self.find_question_by_id(session.current_question_id, questionnaire_def)
        
        # Check for explicit next questions from response option
        option = next((opt for opt in current_question.get("options", []) if opt["value"] == current_response), None)
        if option and option.get("next_questions"):
            # Get first valid next question from option
            for next_q_id in option["next_questions"]:
                next_question = await self.find_question_by_id(next_q_id, questionnaire_def)
                if self.is_question_relevant(next_question, session.context):
                    return next_question
        
        # Check flow-based next question
        current_flow = questionnaire_def["schema"]["flows"].get(session.current_flow, {})
        questions = current_flow.get("questions", [])
        
        # Find current question index in flow
        current_index = next((i for i, q in enumerate(questions) if q["id"] == session.current_question_id), -1)
        
        # Get next question in flow that's relevant to context
        for i in range(current_index + 1, len(questions)):
            next_question = questions[i]
            if self.is_question_relevant(next_question, session.context):
                return next_question
                
        # No more questions in current flow - questionnaire complete
        return None
        
    def is_question_relevant(self, question: Dict, context: Dict[str, Any]) -> bool:
        """Check if question is relevant based on current context"""
        
        # Check skip conditions
        if question.get("skip_conditions"):
            for condition in question["skip_conditions"]:
                if self.evaluate_condition(condition, context):
                    return False
                    
        # Check include conditions  
        if question.get("include_conditions"):
            for condition in question["include_conditions"]:
                if not self.evaluate_condition(condition, context):
                    return False
                    
        return True
        
    async def update_context(self, session: QuestionnaireSession, question_id: str, response: Any, questionnaire_def: Dict) -> Dict[str, Any]:
        """Update session context based on response"""
        
        context_updates = {}
        
        # Get context updates from question option
        current_question = await self.find_question_by_id(question_id, questionnaire_def)
        option = next((opt for opt in current_question.get("options", []) if opt["value"] == response), None)
        
        if option and option.get("context_updates"):
            context_updates.update(option["context_updates"])
            
        # Apply context analysis rules
        analyzed_context = await self.context_analyzer.analyze_response(question_id, response, session.context)
        context_updates.update(analyzed_context)
        
        return context_updates
```

```python
class TemplateManager:
    def __init__(self, db_client):
        self.db = db_client
        
    async def save_as_template(self, session_id: str, template_name: str, description: str, visibility: str = "private") -> str:

            "node_type": session.node_type,

                "average_completion_time": self.calculate_completion_time(session)
            },
            "tags": self.generate_tags(session.context),
            "visibility": visibility,
            "created_by": session.user_id,
            "created_at": datetime.utcnow(),
            "is_active": True
        }
        
        template_id = await self.db.node_templates.insert_one(template)
        return str(template_id.inserted_id)
        
    async def recommend_templates(self, context: Dict[str, Any], node_type: str) -> List[Dict]:
        """Recommend relevant templates based on current context"""
        
        # Build recommendation query
        query = {
            "node_type": node_type,
            "is_active": True,
            "visibility": {"$in": ["public", "organization"]}
        }
        
        # Context-based filtering
        context_filters = []
        if context.get("app_type"):
            context_filters.append({"template_config.context.app_type": context["app_type"]})
        if context.get("complexity_level"):  
            context_filters.append({"template_config.context.complexity_level": context["complexity_level"]})
            
        if context_filters:
            query["$or"] = context_filters
            
        templates = await self.db.node_templates.find(query).sort("usage_stats.usage_count", -1).limit(5).to_list()
        
        # Calculate relevance scores
        for template in templates:
            template["relevance_score"] = self.calculate_relevance(template, context)
            
        return sorted(templates, key=lambda x: x["relevance_score"], reverse=True)
        
    async def apply_template(self, session_id: str, template_id: str) -> Dict[str, Any]:
        """Apply template configuration to current session"""
        
        template = await self.db.node_templates.find_one({"_id": ObjectId(template_id)})
        if not template:
            raise ValueError("Template not found")
            
        session = await self.get_session(session_id)
        
        # Apply template context
        session.context.update(template["template_config"]["context"])
        
        # Apply template responses
        session.responses.update(template["template_config"]["questionnaire_responses"])
        
        # Mark questions as completed
        session.completed_questions.extend(template["template_config"]["questionnaire_responses"].keys())
        session.completed_questions = list(set(session.completed_questions))  # Remove duplicates
        
        # Update progress
        session.progress_percentage = self.calculate_progress(session)
        
        await self.save_session(session)
        
        # Update template usage stats
        await self.db.node_templates.update_one(
            {"_id": ObjectId(template_id)},
            {"$inc": {"usage_stats.usage_count": 1}}
        )
        
        return {
            "applied_context": template["template_config"]["context"],
            "applied_responses": template["template_config"]["questionnaire_responses"],
            "suggested_dependencies": template["template_config"].get("dependency_nodes", []),
            "progress": session.progress_percentage
        }
```

```typescript
// contexts/QuestionnaireSessionContext.tsx
import React, { createContext, useContext, useReducer, useCallback } from 'react';

interface QuestionnaireSession {
  sessionId: string;
  nodeId: string;
  diagramId: string;
  currentFlow: string;
  currentQuestionId: string | null;
  context: Record<string, any>;
  responses: Record<string, any>;
  completedQuestions: string[];
  progressPercentage: number;
  estimatedRemaining: number;
  suggestedDependencies: DependencyNode[];
}

interface QuestionnaireState {
  session: QuestionnaireSession | null;
  currentQuestion: DynamicQuestion | null;
  loading: boolean;
  error: string | null;
  templates: Template[];
}

type QuestionnaireAction =
  | { type: 'START_SESSION'; payload: QuestionnaireSession }
  | { type: 'SET_CURRENT_QUESTION'; payload: DynamicQuestion }
  | { type: 'SUBMIT_RESPONSE'; payload: { questionId: string; response: any } }
  | { type: 'UPDATE_CONTEXT'; payload: Record<string, any> }
  | { type: 'SET_LOADING'; payload: boolean }
  | { type: 'SET_ERROR'; payload: string | null }
  | { type: 'LOAD_TEMPLATES'; payload: Template[] };

const questionnaireReducer = (state: QuestionnaireState, action: QuestionnaireAction): QuestionnaireState => {
  switch (action.type) {
    case 'START_SESSION':
      return {
        ...state,
        session: action.payload,
        loading: false,
        error: null
      };
      
    case 'SET_CURRENT_QUESTION':
      return {
        ...state,
        currentQuestion: action.payload
      };
      
    case 'SUBMIT_RESPONSE':
      if (!state.session) return state;
      return {
        ...state,
        session: {
          ...state.session,
          responses: {
            ...state.session.responses,
            [action.payload.questionId]: action.payload.response
          },
          completedQuestions: [
            ...state.session.completedQuestions,
            action.payload.questionId
          ]
        }
      };
      
    // ... other cases
    default:
      return state;
  }
};

const QuestionnaireSessionContext = createContext<{
  state: QuestionnaireState;
  startSession: (nodeType: string, diagramId: string, nodeId: string) => Promise<void>;
  submitResponse: (questionId: string, response: any) => Promise<void>;
  skipQuestion: () => Promise<void>;
  saveAsTemplate: (name: string, description: string) => Promise<void>;
  applyTemplate: (templateId: string) => Promise<void>;
  acceptDependency: (dependencyId: string) => Promise<void>;
} | null>(null);

export const QuestionnaireSessionProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [state, dispatch] = useReducer(questionnaireReducer, {
    session: null,
    currentQuestion: null,
    loading: false,
    error: null,
    templates: []
  });

  const startSession = useCallback(async (nodeType: string, diagramId: string, nodeId: string) => {
    dispatch({ type: 'SET_LOADING', payload: true });
    
    try {
      const response = await fetch('/api/questionnaire/sessions', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ nodeType, diagramId, nodeId })
      });
      
      const sessionData = await response.json();
      dispatch({ type: 'START_SESSION', payload: sessionData.session });
      
      // Load initial question
      if (sessionData.initialQuestion) {
        dispatch({ type: 'SET_CURRENT_QUESTION', payload: sessionData.initialQuestion });
      }
      
    } catch (error) {
      dispatch({ type: 'SET_ERROR', payload: error.message });
    }
  }, []);

  const submitResponse = useCallback(async (questionId: string, response: any) => {
    if (!state.session) return;
    
    dispatch({ type: 'SET_LOADING', payload: true });
    dispatch({ type: 'SUBMIT_RESPONSE', payload: { questionId, response } });
    
    try {
      const apiResponse = await fetch(`/api/questionnaire/sessions/${state.session.sessionId}/respond`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ questionId, response })
      });
      
      const responseData = await apiResponse.json();
      
      // Update context
      if (responseData.contextUpdates) {
        dispatch({ type: 'UPDATE_CONTEXT', payload: responseData.contextUpdates });
      }
      
      // Set next question or complete
      if (responseData.nextQuestion) {
        dispatch({ type: 'SET_CURRENT_QUESTION', payload: responseData.nextQuestion });
      } else {
        // Questionnaire complete
        dispatch({ type: 'SET_CURRENT_QUESTION', payload: null });
      }
      
    } catch (error) {
      dispatch({ type: 'SET_ERROR', payload: error.message });
    } finally {
      dispatch({ type: 'SET_LOADING', payload: false });
    }
  }, [state.session]);

  return (
    <QuestionnaireSessionContext.Provider value={{
      state,
      startSession,
      submitResponse,
      skipQuestion,
      saveAsTemplate,
      applyTemplate,
      acceptDependency
    }}>
      {children}
    </QuestionnaireSessionContext.Provider>
  );
};

export const useQuestionnaireSession = () => {
  const context = useContext(QuestionnaireSessionContext);
  if (!context) {
    throw new Error('useQuestionnaireSession must be used within QuestionnaireSessionProvider');
  }
  return context;
};
```

```python
class QuestionResponseValidator:
    @staticmethod
    def validate_response(question: DynamicQuestion, response: Any) -> ValidationResult:
        # Type validation
        if question.type == "single_choice":
            valid_values = [opt.value for opt in question.options]
            if response not in valid_values:
                return ValidationResult(valid=False, error="Invalid option selected")
                
        # Custom validation rules
        if question.validation_rules:
            for rule_name, rule_config in question.validation_rules.items():
                if not validate_rule(rule_name, response, rule_config):

        return ValidationResult(valid=True)
```

#### **2. Session Security**
```python
class SecureSessionManager:
    def __init__(self, redis_client):
        self.redis = redis_client
        self.session_timeout = 3600  # 1 hour
        
    async def create_session(self, user_id: str, diagram_id: str) -> str:
        session_id = secrets.token_urlsafe(32)
        session_key = f"questionnaire_session:{session_id}"
        
        session_data = {
            "user_id": user_id,
            "diagram_id": diagram_id,
            "created_at": datetime.utcnow().isoformat(),
            "last_activity": datetime.utcnow().isoformat()
        }
        
        await self.redis.setex(session_key, self.session_timeout, json.dumps(session_data))
        return session_id
        
    async def validate_session(self, session_id: str, user_id: str) -> bool:
        session_key = f"questionnaire_session:{session_id}"
        session_data = await self.redis.get(session_key)
        
        if not session_data:
            return False
            
        data = json.loads(session_data)
        return data.get("user_id") == user_id
```

```python
class QuestionFlowCache:
    def __init__(self, redis_client):
        self.redis = redis_client
        self.cache_ttl = 300  # 5 minutes
        
    async def get_next_questions(self, context_hash: str) -> Optional[List[DynamicQuestion]]:
        cache_key = f"question_flow:{context_hash}"
        cached_data = await self.redis.get(cache_key)
        
        if cached_data:
            return [DynamicQuestion(**q) for q in json.loads(cached_data)]
        return None
        
    async def cache_question_flow(self, context_hash: str, questions: List[DynamicQuestion]):
        cache_key = f"question_flow:{context_hash}"
        question_data = [q.dict() for q in questions]
        await self.redis.setex(cache_key, self.cache_ttl, json.dumps(question_data))
```

```python
class OptimizedQuestionnaireRepository:
```
